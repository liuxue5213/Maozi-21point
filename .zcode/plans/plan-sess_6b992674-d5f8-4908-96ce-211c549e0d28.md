# 账户体系和积分继承系统实现计划

## 📋 项目概述
为帽子21点游戏添加完整的账户体系，包括用户认证、积分继承、社交功能、排行榜、成就系统和管理员后台。

## 🗄️ 第一阶段：数据库设计

### 1.1 数据库选择：SQLite
- 轻量级文件数据库，无需独立服务器
- 适合中小规模游戏
- 零配置，部署简单

### 1.2 数据库表结构

```sql
-- 用户表
CREATE TABLE users (
  id TEXT PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  chips INTEGER DEFAULT 1000,
  avatar TEXT DEFAULT 'default',
  level INTEGER DEFAULT 1,
  experience INTEGER DEFAULT 0,
  games_played INTEGER DEFAULT 0,
  games_won INTEGER DEFAULT 0,
  total_winnings INTEGER DEFAULT 0,
  status TEXT DEFAULT 'active', -- active, banned, muted
  role TEXT DEFAULT 'user', -- user, admin, moderator
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  last_login DATETIME
);

-- 游戏历史表
CREATE TABLE game_history (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id TEXT,
  game_mode TEXT, -- 'pve' or 'pvp'
  result TEXT, -- 'win', 'lose', 'push', 'blackjack'
  chips_change INTEGER,
  opponent_name TEXT,
  duration_seconds INTEGER,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 好友关系表
CREATE TABLE friendships (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id TEXT,
  friend_id TEXT,
  status TEXT DEFAULT 'pending', -- pending, accepted, blocked
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (friend_id) REFERENCES users(id),
  UNIQUE(user_id, friend_id)
);

-- 成就表
CREATE TABLE achievements (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT,
  icon TEXT,
  requirement_type TEXT, -- 'games_played', 'games_won', 'chips_earned', 'streak'
  requirement_value INTEGER,
  reward_chips INTEGER DEFAULT 0,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 用户成就关联表
CREATE TABLE user_achievements (
  user_id TEXT,
  achievement_id TEXT,
  progress INTEGER DEFAULT 0,
  completed BOOLEAN DEFAULT 0,
  completed_at DATETIME,
  PRIMARY KEY (user_id, achievement_id),
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (achievement_id) REFERENCES achievements(id)
);

-- 管理员操作日志
CREATE TABLE admin_logs (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  admin_id TEXT,
  action TEXT,
  target_id TEXT,
  details TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (admin_id) REFERENCES users(id)
);
```

### 2.2 Socket.IO认证集成

```javascript
// 在Socket连接时验证JWT
io.use((socket, next) => {
  const token = socket.handshake.auth.token;
  if (!token) return next(new Error('认证失败'));
  
  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    socket.userId = decoded.id;
    next();
  } catch (err) {
    next(new Error('Token无效'));
  }
});
```
